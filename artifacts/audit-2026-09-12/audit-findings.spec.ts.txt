import { test, expect } from '@playwright/test';
test.beforeEach(({}, info) => test.skip(info.project.name !== 'chromium-wide-1440', 'Reprodução direcionada desktop'));

test('audit: paid charge is included in open total', async ({page}) => {
  await page.route('**/api/v1/finance/charges', route => route.fulfill({json: {schemaVersion:1,correlationId:'audit-finance',data:{items:[{id:'paid-audit',description:'Cobrança já paga',amountCents:10000,status:'PAID',createdAt:'2026-09-12T12:00:00Z'}]}}}));
  await page.goto('/');
  await page.getByRole('button',{name:/Abrir demonstração sintética/i}).click();
  await expect(page.getByRole('heading',{name:'Bom dia, Ricardo.'})).toBeVisible();
  await page.getByRole('button',{name:'Financeiro',exact:true}).click();
  await expect(page.locator('.finance-summary')).toContainText('100,00');
  await expect(page.locator('.finance-summary')).toContainText('EM ABERTO');
  await expect(page.getByText('Pago',{exact:true})).toBeVisible();
});

test('audit: logout failure leaves session usable after reload', async ({page}) => {
  await page.goto('/');
  await page.getByRole('button',{name:/Abrir demonstração sintética/i}).click();
  await expect(page.getByRole('heading',{name:'Bom dia, Ricardo.'})).toBeVisible();
  await page.route('**/api/v1/auth/logout', route => route.abort('failed'));
  await page.getByRole('button',{name:/Sair/i}).click();
  await expect(page.getByRole('button',{name:/Abrir demonstração sintética/i})).toBeVisible();
  await page.unroute('**/api/v1/auth/logout');
  await page.reload();
  await expect(page.getByRole('heading',{name:'Bom dia, Ricardo.'})).toBeVisible();
});
